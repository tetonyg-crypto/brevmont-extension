// =============================================================================
// PASTE THIS INTO: Chrome → chrome://extensions → Brevmont → "Service worker"
//   (or "Inspect views: service worker") → Console
//
// This bypasses the onboarding wizard by setting the same keys the wizard
// writes. You CANNOT run this from Node or a normal web page — only inside
// the extension service worker (or an extension page), because only there
// does `chrome.storage` apply to the extension.
//
// LIMIT: If you only set the HUMAN license key (ABDK-...) as dealer_token,
// signed API calls may return 401 — the real flow stores a server-issued
// dtk_ UUID in storage after /v1/license/validate. This bypass is still useful
// to open the pill/sidebar; full generation may need BREZ’s token fix.
// =============================================================================

(() => {
  const LICENSE_OR_DTK = 'ABDK-PTUQ-HPW4'; // Replace with dtk_... if you have it from API
  const repName = 'Garcia Owner';
  const dealership = 'Garcia Test Motors';

  const localPayload = {
    dealer_token: LICENSE_OR_DTK,
    brevmont_tone: 'professional',
    brevmont_goal: 'close_deal',
    profile_onboarded: true,
    brevmont_dealership_id: 'de460133-b420-4880-bcfa-4023de2f56b3',
    brevmont_user_id: 'f3615c3b-b825-4240-a0a3-c869ad5e058c',
    brevmont_role: 'owner',
    dealership_id: 'de460133-b420-4880-bcfa-4023de2f56b3',
  };

  const syncPayload = {
    profile_onboarded: true,
    dealer_token: LICENSE_OR_DTK,
    rep_name: repName,
    dealership: dealership,
    dealership_id: 'de460133-b420-4880-bcfa-4023de2f56b3',
  };

  chrome.storage.local.set(localPayload, () => {
    if (chrome.runtime.lastError) console.error('local', chrome.runtime.lastError);
    else console.log('chrome.storage.local OK', Object.keys(localPayload));
  });
  chrome.storage.sync.set(syncPayload, () => {
    if (chrome.runtime.lastError) console.error('sync', chrome.runtime.lastError);
    else console.log('chrome.storage.sync OK', Object.keys(syncPayload));
  });
})();
